PROGRAM PLC_PRG
VAR
	Delay: TON;
	LedIn AT %IB32 : BOOL;
	LedOut AT %QB32 : BOOL;
	// mode: 4 - AFE digital I/O's
	a1o_mode AT %QD9 : UDINT;
	// digital i/o afe#1,8/J19
	d1i AT %IB0 : BOOL;
	// when configured as input, this field is unused
	d1o AT %QB0 : BOOL;
	// pin1 mode: FALSE - input; TRUE - output
	d25o_d1mode AT %QB24 : BOOL;
	// digital i/o afe#2,7/J19
	d2i AT %IB1 : BOOL;
	// when configured as input, this field is unused
	d2o AT %QB1 : BOOL;
	// pin2 mode: FALSE - input; TRUE - output
	d26o_d2mode AT %QB25 : BOOL;
	// digital i/o afe#3,6/J19
	d3i AT %IB2 : BOOL;
	// when configured as input, this field is unused
	d3o AT %QB2 : BOOL;
	// pin3 mode: FALSE - input; TRUE - output
	d27o_d3mode AT %QB26 : BOOL;
	// digital i/o afe#4,5/J19
	d4i AT %IB3 : BOOL;
	// when configured as input, this field is unused
	d4o AT %QB3 : BOOL;
	// pin4 mode: FALSE - input; TRUE - output
	d28o_d4mode AT %QB27 : BOOL;
	// digital i/o afe#5,4/J19
	d5i AT %IB4 : BOOL;
	// when configured as input, this field is unused
	d5o AT %QB4 : BOOL;
	// pin5 mode: FALSE - input; TRUE - output
	d29o_d5mode AT %QB28 : BOOL := TRUE;
	// digital i/o afe#6,3/J19
	d6i AT %IB5 : BOOL;
	// when configured as input, this field is unused
	d6o AT %QB5 : BOOL;
	// pin6 mode: FALSE - input; TRUE - output
	d30o_d6mode AT %QB29 : BOOL := TRUE;
	// digital i/o afe#7,2/J19
	d7i AT %IB6 : BOOL;
	// when configured as input, this field is unused
	d7o AT %QB6 : BOOL;
	// pin7 mode: FALSE - input; TRUE - output
	d31o_d7mode AT %QB30 : BOOL := TRUE;
	// digital i/o afe#8,1/J19
	d8i AT %IB7 : BOOL;
	// when configured as input, this field is unused
	d8o AT %QB7 : BOOL;
	// pin8 mode: FALSE - input; TRUE - output
	d32o_d8mode AT %QB31 : BOOL := TRUE;
END_VAR


Delay(IN:=TRUE, PT:=T#1S);
IF Delay.Q THEN
	a1o_mode := 4;
	LedOut:= NOT LedIn;
	//d8o:= NOT d8i;
	Delay(IN:=FALSE);
END_IF;
